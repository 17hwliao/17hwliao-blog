static class Test implements Iterable<String>{
    @Override
    public Iterator<String> iterator() {
        return new Iterator<String>(){   //生成匿名的Iterator对象
            @Override
            public boolean hasNext() {
                return true ;
            }
            @Override
            public String next(){
                return "gay" ;
            }
        } ;
    }
}
