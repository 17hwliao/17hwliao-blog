    long test(int x ){
        if(x <= 0 ) return 0 ;
        if(x  < 3 ) return 1 ;
        long [] dp = new long [ x + 1 ];
        dp[0] = 0 ;
        dp[1] = 1 ;
        dp[2] = 1 ;
        for(int i = 3 ; i <= x ; i ++){
            dp[i] = dp[i - 1] + dp[i -2 ] ;
        }
        return dp[x] ;
    }
