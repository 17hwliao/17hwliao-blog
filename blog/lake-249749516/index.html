package com_java;

public class Person {
     String name ;
     int age;

    public Person(String ahs, int i) {
        name = ahs;
        age = i;
    }

    public Person() {
        String name ;
        int age ;
    }

    public void setName( String name ) {
        //封装name
        this. name = name ;
    }
    public void setAge(int age) {
        //封装age
        this.age = age;
    }
    public int getAge() {
        return age;
    }
    public String getName() {
        return name;
    }

    public static Person new_i(){
        //实现封装创建
        return new Person() ;
    }
}

package com_java;

public class Student extends Person {

    String name;

    public Student(){
        super("ahs" , 11) ;
    }

    public void study(){
        System.out.println(super.name + age);
    }

}
